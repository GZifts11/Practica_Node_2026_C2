-- CREAR TABLAS

CREATE TABLE Materias (
ID INT IDENTITY (1,1) PRIMARY KEY,
NOMBRE VARCHAR(50) NOT NULL
);

CREATE TABLE Turnos  (
ID INT IDENTITY (1,1) PRIMARY KEY,
NOMBRE VARCHAR(50) NOT NULL
);

CREATE TABLE Comisiones  (
ID INT IDENTITY (1,1) PRIMARY KEY,
NOMBRE CHAR NOT NULL
);

CREATE TABLE Lenguajes (
ID INT IDENTITY (1,1) PRIMARY KEY,
MATERIA_ID INT NOT NULL,
NOMBRE VARCHAR(50) NOT NULL,
TURNO_ID INT NOT NULL,
COMISION_ID INT NOT NULL,
CANTIDAD_ALUMNOS INT NOT NULL,
FOREIGN KEY (MATERIA_ID) REFERENCES Materias(ID),
FOREIGN KEY (TURNO_ID) REFERENCES Turnos(ID),
FOREIGN KEY (COMISION_ID) REFERENCES Comisiones(ID)
)

GO

-- CONSULTAS

SELECT * FROM Lenguajes

GO

SELECT
l.ID,
m.NOMBRE AS MATERIA,
l.NOMBRE,
t.NOMBRE AS TURNO,
c.NOMBRE AS COMISION ,
l.CANTIDAD_ALUMNOS
FROM Lenguajes l
JOIN
TURNOS t ON l.TURNO_ID = t.ID
JOIN
MATERIAS m ON l.MATERIA_ID = m.ID
JOIN
COMISIONES c ON l.COMISION_ID = c.ID

GO
